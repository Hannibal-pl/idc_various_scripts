Some simple idc scripts I've used.

analyze_id0.idc:
 -- Script analyzing ida id0 database.

copy_data.idc
 -- Script copying data from source to destination address

dump_to_file.idc:
 -- Script dumping some part of database to file.

patch_databse.idc:
 -- Script patching database at selected address using data from selected file.

t1_decode.idc
 -- Script analyzing Type 1/2 font data

unpack_lz77_firmware.idc:
 -- Script unpacking lz77 like data stream found in some firmware.

zero_unpack_data.idc:
 -- Script unpacking zero RLE like data stream found in some firmware.
